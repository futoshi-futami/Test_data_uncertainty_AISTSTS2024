                "id": int(taxon.get("id") or -1),
                "rank": rank,
                "name": str(taxon.get("name") or f"unknown-{rank}"),
            }
    missing = [r for r in TAXONOMIC_RANK_ORDER if r not in rank_to_taxon]
    if missing:
        raise RuntimeError(f"taxonomy is missing ranks {missing} for {detail.get('name')}")
    return [rank_to_taxon[r] for r in TAXONOMIC_RANK_ORDER]


def download_one_image(record: Mapping[str, Any], destination: Path, timeout: int = 60) -> Tuple[bool, str]:
    if destination.exists() and destination.stat().st_size > 500:
        try:
            with Image.open(destination) as image:
                image.verify()
            return True, "cached"
        except Exception:
            destination.unlink(missing_ok=True)
    session = requests.Session()
    session.headers.update({"User-Agent": "HUC-composition-experiment/1.0"})
    last = ""
    for attempt in range(6):
        try:
            response = session.get(str(record["photo_url"]), timeout=timeout)
            response.raise_for_status()
            tmp = destination.with_suffix(destination.suffix + ".part")
            tmp.parent.mkdir(parents=True, exist_ok=True)
            tmp.write_bytes(response.content)
            with Image.open(tmp) as image:
                image.convert("RGB").load()
            tmp.replace(destination)
            return True, "downloaded"
        except Exception as exc:
            last = repr(exc)
            time.sleep(min(10, 1.5 ** attempt))
    return False, last


def acquire_real_images(root: Path, images_per_species: int) -> Tuple[pd.DataFrame, List[List[Dict[str, Any]]]]:
    data_dir = root / "data"
    image_dir = data_dir / "images"
    data_dir.mkdir(parents=True, exist_ok=True)
    api = INatAPI(data_dir / "api_cache")

    selected_taxa: List[Dict[str, Any]] = []
    for genus_name, count in GENUS_TARGETS:
        genus = api.exact_genus(genus_name)
        species = api.species_children(genus, count, minimum_observations=max(500, images_per_species * 4))
        for row in species:
            detail = api.taxon_detail(int(row["id"]))
            path = normalized_taxonomy_path(detail)
            selected_taxa.append({
                "genus_target": genus_name,
                "taxon_id": int(detail["id"]),
                "scientific_name": str(detail["name"]),
                "observations_count": int(detail.get("observations_count") or 0),
                "taxonomy": path,
            })

    selected_taxa.sort(key=lambda x: (next(i for i, z in enumerate(GENUS_TARGETS) if z[0] == x["genus_target"]), x["scientific_name"]))
    if len(selected_taxa) != sum(x[1] for x in GENUS_TARGETS):
        raise AssertionError("selected species count mismatch")

    download_jobs: List[Tuple[Dict[str, Any], Path, Dict[str, Any]]] = []
    oversample = max(images_per_species + 30, int(math.ceil(images_per_species * 1.35)))
    for class_index, species in enumerate(selected_taxa):
        candidates = api.observation_candidates(int(species["taxon_id"]), oversample)
        if len(candidates) < images_per_species:
            raise RuntimeError(f"only {len(candidates)} observations for {species['scientific_name']}")
        species_dir = image_dir / f"{class_index:02d}_{species['scientific_name'].replace(' ', '_')}"
        for candidate in candidates:
            destination = species_dir / f"obs_{candidate['observation_id']}_photo_{candidate['photo_id']}.jpg"
            download_jobs.append((candidate, destination, species))

    print(f"Downloading/validating {len(download_jobs)} candidate images", flush=True)
    status: Dict[str, Tuple[bool, str]] = {}
    with ThreadPoolExecutor(max_workers=16) as executor:
        futures = {
            executor.submit(download_one_image, rec, path): str(path)
            for rec, path, _ in download_jobs
        }
        completed = 0
        for future in as_completed(futures):
            status[futures[future]] = future.result()
            completed += 1
            if completed % 250 == 0:
                print(f"  image jobs {completed}/{len(download_jobs)}", flush=True)

    manifest_rows: List[Dict[str, Any]] = []
    taxonomy_paths: List[List[Dict[str, Any]]] = []
    for class_index, species in enumerate(selected_taxa):
        taxonomy_paths.append(species["taxonomy"])
        candidates_for_species: List[Tuple[int, Dict[str, Any], Path]] = []
        for candidate, destination, spec in download_jobs:
            if int(spec["taxon_id"]) != int(species["taxon_id"]):
                continue
            ok, message = status.get(str(destination), (False, "missing status"))
            if ok:
                candidates_for_species.append((int(candidate["observation_id"]), candidate, destination))
        candidates_for_species.sort(key=lambda x: x[0])
        if len(candidates_for_species) < images_per_species:
            raise RuntimeError(
                f"only {len(candidates_for_species)} valid images for {species['scientific_name']}, need {images_per_species}"
            )
        for _, candidate, destination in candidates_for_species[:images_per_species]:
            manifest_rows.append({
                "class_index": class_index,
                "scientific_name": species["scientific_name"],
                "taxon_id": species["taxon_id"],
                "genus_target": species["genus_target"],
                "observation_id": candidate["observation_id"],
                "photo_id": candidate["photo_id"],
                "photo_url": candidate["photo_url"],
                "photo_license": candidate["photo_license"],
                "observed_on": candidate["observed_on"],
                "place_guess": candidate["place_guess"],
                "local_path": str(destination),
                "observations_count_species": species["observations_count"],
                "taxonomy_json": json.dumps(species["taxonomy"], ensure_ascii=False),
            })

    manifest = pd.DataFrame(manifest_rows).sort_values(["class_index", "observation_id"]).reset_index(drop=True)
    expected = images_per_species * len(selected_taxa)
    if len(manifest) != expected or manifest.groupby("class_index").size().nunique() != 1:
        raise AssertionError((len(manifest), expected, manifest.groupby("class_index").size().to_dict()))
    manifest.to_csv(data_dir / "image_manifest.csv", index=False)
    (data_dir / "selected_species.json").write_text(
        json.dumps(selected_taxa, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    return manifest, taxonomy_paths


class ImageDataset:
    def __init__(self, paths: Sequence[str], transform: Any) -> None:
        self.paths = list(paths)
        self.transform = transform

    def __len__(self) -> int:
        return len(self.paths)

    def __getitem__(self, index: int) -> Tuple[Any, int, float, float, float]:
        path = self.paths[index]
        with Image.open(path) as image:
            image = image.convert("RGB")
            width, height = image.size
            small = image.resize((64, 64))
            gray = np.asarray(small.convert("L"), dtype=np.float64) / 255.0
            brightness = float(gray.mean())
            contrast = float(gray.std())
            aspect = float(width / max(height, 1))
            tensor = self.transform(image)
        return tensor, index, brightness, contrast, aspect


def extract_resnet18_features(root: Path, manifest: pd.DataFrame) -> Dict[str, np.ndarray]:
    cache = root / "data" / "resnet18_features.npz"
    if cache.exists():
        loaded = np.load(cache, allow_pickle=True)
        result = {k: loaded[k] for k in loaded.files}
        if len(result["features"]) == len(manifest):
            return result

    import torch
    from torch.utils.data import DataLoader
    from torchvision.models import ResNet18_Weights, resnet18

    torch.set_num_threads(max(1, min(2, os.cpu_count() or 1)))
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    weights = ResNet18_Weights.DEFAULT
    model = resnet18(weights=weights)
    model.fc = torch.nn.Identity()
    model.eval().to(device)
    dataset = ImageDataset(manifest["local_path"].tolist(), weights.transforms())
    loader = DataLoader(dataset, batch_size=64 if device.type == "cuda" else 32, shuffle=False, num_workers=2)

    features = np.empty((len(dataset), 512), dtype=np.float32)
    brightness = np.empty(len(dataset), dtype=np.float32)
    contrast = np.empty(len(dataset), dtype=np.float32)
    aspect = np.empty(len(dataset), dtype=np.float32)
    print(f"Extracting ResNet-18 features on {device}", flush=True)
    with torch.no_grad():
        for batch_index, (images, indices, b, c, a) in enumerate(loader):
            output = model(images.to(device)).detach().cpu().numpy().astype(np.float32)
            idx = indices.numpy()
            features[idx] = output
            brightness[idx] = b.numpy().astype(np.float32)
            contrast[idx] = c.numpy().astype(np.float32)
            aspect[idx] = a.numpy().astype(np.float32)
            if (batch_index + 1) % 20 == 0:
                print(f"  feature batches {batch_index + 1}/{len(loader)}", flush=True)

    result = {
        "features": features,
        "labels": manifest["class_index"].to_numpy(dtype=np.int64),
        "brightness": brightness,
        "contrast": contrast,
        "aspect": aspect,
        "paths": manifest["local_path"].to_numpy(dtype=object),
        "scientific_names": manifest["scientific_name"].to_numpy(dtype=object),
        "observation_ids": manifest["observation_id"].to_numpy(dtype=np.int64),
    }
    np.savez_compressed(cache, **result)
    return result


@dataclass
class Node:
    index: int
    name: str
    rank: str
    parent: int
    children: List[int]
    depth: int
    path_key: str


class TaxonomyTree:
    def __init__(self, taxonomy_paths: Sequence[Sequence[Mapping[str, Any]]]) -> None:
        self.nodes: List[Node] = [Node(0, "Life", "root", -1, [], 0, "Life")]
        key_to_index: Dict[Tuple[Tuple[str, str], ...], int] = {tuple(): 0}
        self.leaf_nodes: List[int] = []
        self.class_names: List[str] = []
        for class_index, path in enumerate(taxonomy_paths):
            prefix: List[Tuple[str, str]] = []
            parent = 0
            for taxon in path:
                rank = str(taxon["rank"])
                name = str(taxon["name"])
                prefix.append((rank, name))
                key = tuple(prefix)
                if key not in key_to_index:
                    idx = len(self.nodes)
                    self.nodes.append(Node(idx, name, rank, parent, [], self.nodes[parent].depth + 1, "/".join(f"{r}:{n}" for r, n in prefix)))
                    self.nodes[parent].children.append(idx)
                    key_to_index[key] = idx
                parent = key_to_index[key]
            self.leaf_nodes.append(parent)
            self.class_names.append(str(path[-1]["name"]))
        self.num_nodes = len(self.nodes)
        self.num_leaves = len(self.leaf_nodes)
        self.rank_nodes: Dict[str, List[int]] = {rank: [] for rank in TAXONOMIC_RANK_ORDER}
        for node in self.nodes:
            if node.rank in self.rank_nodes:
                self.rank_nodes[node.rank].append(node.index)
        self.internal_nodes = [node.index for node in self.nodes if node.children]
        self.internal_position = {node: j for j, node in enumerate(self.internal_nodes)}
        self.leaf_mask = np.zeros((self.num_nodes, self.num_leaves), dtype=np.float64)
        for leaf_index, node_index in enumerate(self.leaf_nodes):
            current = node_index
            while current >= 0:
                self.leaf_mask[current, leaf_index] = 1.0
                current = self.nodes[current].parent
        self.path_cache: Dict[int, List[int]] = {}
        for node in range(self.num_nodes):
            path_nodes: List[int] = []
            current = node
            while current >= 0:
                path_nodes.append(current)
                current = self.nodes[current].parent
            self.path_cache[node] = list(reversed(path_nodes))
